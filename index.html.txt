<!-- Floating 💞 particles script -->
<script>
const nameContainer = document.getElementById('nameParticles');
const nameCount = 40;
const colors = ['#ff69b4', '#ff1493', '#ff00ff', '#ffb6c1']; // bright pink colors

for(let i=0;i<nameCount;i++){
    const n = document.createElement('div');
    n.className = 'name-particle';
    const size = Math.random()*1 + 0.7;
    const opacity = Math.random()*0.5 + 0.3;
    const drift = (Math.random()-0.5)*200;
    const duration = 10 + Math.random()*10;
    n.style.setProperty('--size', size);
    n.style.setProperty('--opacity', opacity);
    n.style.setProperty('--drift', drift+'px');
    n.style.fontSize = (Math.random()*16 + 12)+'px';
    n.style.left = Math.random()*100+'vw';
    n.style.top = Math.random()*100+'vh';
    const color = colors[Math.floor(Math.random()*colors.length)];
    n.style.color = color;
    n.style.textShadow = `0 0 2px ${color}, 0 0 5px ${color}`; // subtle glow
    n.textContent = "💞";
    n.style.animationDuration = duration+'s';
    nameContainer.appendChild(n);
}
</script>
